64void vibrar(void *puntero_tarea_vibrador){
65 uint16_t tiempo_ON = 0;
66 uint16_t tiempo_OFF = 0;
67
68 while (true){
69 if (!encendido){
70 GPIOOff(GPIO_9);
71 LedOff(LED_1);
72 LedOff(LED_2);
73 LedOff(LED_3);
74 vTaskDelay(200 / portTICK_PERIOD_MS);
75 continue;
76 }
77
78 // Rango de distancia
79 if (distancia > DIST_ALTO){
80 // Muy lejos → no vibra
81 GPIOOff(GPIO_9);
82 LedOff(LED_1);
83 LedOff(LED_2);
84 LedOff(LED_3);
85 tiempo_OFF = 700;
86 }
87
88 else if (distancia <= DIST_ALTO && distancia > DIST_MEDIA){
89 LedOn(LED_3);
90 LedOff(LED_2);
91 LedOff(LED_1);
92 tiempo_ON = 100;
93 tiempo_OFF = 800;
94 }
95 else if (distancia <= DIST_MEDIA && distancia >= DIST_MIN){
96 LedOn(LED_2);
97 LedOff(LED_1);
98 LedOff(LED_3);
99 tiempo_ON = 150;
100 tiempo_OFF = 500;
101 }
102 else if (distancia < DIST_MIN){
103 LedOn(LED_1);
104 LedOff(LED_2);
105 LedOff(LED_3);
106 tiempo_ON = 200;
107 tiempo_OFF = 100;
108 }
109
110 // Vibración tipo pulso
111 if (distancia <= DIST_ALTO){
112 GPIOOn(GPIO_9);
113 vTaskDelay(tiempo_ON / portTICK_PERIOD_MS);
114 GPIOOff(GPIO_9);
115 }
116
117 vTaskDelay(tiempo_OFF / portTICK_PERIOD_MS);
118 }
119}
